async function initAudio() { try { if (!audioCtx) { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } if (audioCtx.state !== "running") { await audioCtx.resume(); } } catch (e) { console.error("Audio initialization failed:", e); } } let jumpscareOsc = null; let jumpscareGain = null; async function startJumpscareSound() { try { if (!audioCtx) { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } if (audioCtx.state !== "running") { await audioCtx.resume(); } const now = audioCtx.currentTime; // Master volume jumpscareGain = audioCtx.createGain(); jumpscareGain.gain.setValueAtTime(1.0, now); jumpscareGain.connect(audioCtx.destination); // Distortion const distortion = audioCtx.createWaveShaper(); const samples = 44100; const curve = new Float32Array(samples); for (let i = 0; i < samples; i++) { const x = (i * 2) / samples - 1; curve[i] = Math.tanh(x * 4); } distortion.curve = curve; distortion.oversample = "4x"; // High-pitched scream const osc1 = audioCtx.createOscillator(); osc1.type = "sawtooth"; osc1.frequency.setValueAtTime(2500, now); osc1.frequency.exponentialRampToValueAtTime(700, now + 3.5); // Second scream layer const osc2 = audioCtx.createOscillator(); osc2.type = "square"; osc2.frequency.setValueAtTime(3200, now); osc2.frequency.exponentialRampToValueAtTime(900, now + 3.5); // Noise const bufferSize = Math.floor(audioCtx.sampleRate * 3.5); const buffer = audioCtx.createBuffer( 1, bufferSize, audioCtx.sampleRate ); const data = buffer.getChannelData(0); for (let i = 0; i < bufferSize; i++) { data[i] = Math.random() * 2 - 1; } const noise = audioCtx.createBufferSource(); noise.buffer = buffer; // Individual volume controls const osc1Gain = audioCtx.createGain(); const osc2Gain = audioCtx.createGain(); const noiseGain = audioCtx.createGain(); osc1Gain.gain.value = 0.35; osc2Gain.gain.value = 0.25; noiseGain.gain.value = 0.12; osc1.connect(osc1Gain); osc2.connect(osc2Gain); noise.connect(noiseGain); osc1Gain.connect(distortion); osc2Gain.connect(distortion); noiseGain.connect(distortion); distortion.connect(jumpscareGain); osc1.start(now); osc2.start(now); noise.start(now); osc1.stop(now + 3.5); osc2.stop(now + 3.5); noise.stop(now + 3.5); jumpscareOsc = { osc1, osc2, noise }; } catch (e) { console.error("Jumpscare audio failed:", e); } } function stopJumpscareSound() { if (jumpscareOsc) { try { jumpscareOsc.osc1.stop(); } catch (e) {} try { jumpscareOsc.osc2.stop(); } catch (e) {} try { jumpscareOsc.noise.stop(); } catch (e) {} jumpscareOsc = null; } if (jumpscareGain) { try { jumpscareGain.disconnect(); } catch (e) {} jumpscareGain = null; } } async function startJumpscare() { state = "scare"; scare_start = Date.now(); introSound.pause(); introSound.currentTime = 0; await startJumpscareSound(); } async function handleInput(x, y) { await initAudio();
